// Updated Smiley Tally: Excludes Limited or Low entries function getSelfControlCountsForDay(name, day, classType = 'regular') { let counts = { high: 0, limited: 0, low: 0, total: 0 }; // 1. If absent for the whole day -> 0 smileys if (db.absentDay[day]?.[name]) { return counts; } let blocks = []; const isRegularStudent = (db.regularNames || []).includes(name); if (classType === 'maths') { blocks = ['Maths']; } else { blocks = isRegularStudent ? ['Block 1', 'Block 2', 'Block 3', 'Maths'] : ['Maths']; } let submittedCount = 0; let highCount = 0; let hasIssues = false; blocks.forEach(blk => { const refKey = `${day}-${blk}`; if (isBlockSubmitted(refKey)) { submittedCount++; const sc = getEffectiveSC(refKey, name, day); if (sc === 'high') highCount++; if (sc === 'limited') { counts.limited++; hasIssues = true; // Exclude smiley if limited is present } if (sc === 'low') { counts.low++; hasIssues = true; // Exclude smiley if low is present } } }); // Award MAX 1 Smiley for the day ONLY if present, submitted, and NO limited/low clicked if (submittedCount > 0 && highCount > 0 && !hasIssues) { counts.high = 1; counts.total = 1; } return counts; } // Fixed Low Click Cycle: // Click 1 = 15m Timer | Click 2 = Private Practice at Break | Click 3 = Reset function handleLowClick(name) { const key = getKey(); if (!db.reflections[key]) db.reflections[key] = {}; if (!db.reflections[key][name]) db.reflections[key][name] = {}; const data = db.reflections[key][name]; // If transitioning from non-low rating, start clean at step 0 if (data.sc !== 'low') { data.lowStage = 0; } // Advance to next stage (1 -> 2 -> 0) let nextStage = ((data.lowStage || 0) % 3) + 1; if (nextStage === 1) { // First click: Start 15-minute Timer data.sc = 'low'; data.lowStage = 1; data.lowTimerStart = Date.now(); delete data.returningAtBreak; } else if (nextStage === 2) { // Second click: Switch directly to Private Practice at Break data.sc = 'low'; data.lowStage = 2; data.returningAtBreak = true; delete data.lowTimerStart; } else { // Third click: Clear rating back to unselected data.sc = null; data.lowStage = 0; delete data.lowTimerStart; delete data.returningAtBreak; } save(); }
